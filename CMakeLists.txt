cmake_minimum_required(VERSION 3.22)
project(VirtualEmbeddedLab VERSION 0.1.0 LANGUAGES C ASM)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
option(VEL_ARM "Build Cortex-M3 firmware" OFF)
add_compile_options(-Wall -Wextra -Werror -Wpedantic -Wconversion -Wshadow)
include(CTest)
option(VEL_FREERTOS "Build independent FreeRTOS ARM firmware" OFF)
if(VEL_FREERTOS)
  include(cmake/stage3.cmake)
elseif(VEL_ARM)
  add_executable(cortex_m3_demo platform/qemu_mps2/startup/startup.S apps/cortex_m3_demo/main.c platform/qemu_mps2/drivers/uart.c)
  target_include_directories(cortex_m3_demo PRIVATE platform/qemu_mps2/drivers)
  target_compile_options(cortex_m3_demo PRIVATE -mcpu=cortex-m3 -mthumb -Wa,--noexecstack -ffreestanding -fno-builtin -ffunction-sections -fdata-sections)
  target_link_options(cortex_m3_demo PRIVATE -mcpu=cortex-m3 -mthumb -nostdlib -T${CMAKE_SOURCE_DIR}/platform/qemu_mps2/linker/mps2.ld -Wl,--gc-sections,--fatal-warnings,-Map=${CMAKE_BINARY_DIR}/cortex_m3_demo.map)
  set_target_properties(cortex_m3_demo PROPERTIES SUFFIX ".elf" LINK_DEPENDS "${CMAKE_SOURCE_DIR}/platform/qemu_mps2/linker/mps2.ld")
  if(BUILD_TESTING)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    foreach(case IN ITEMS boot_uart_ticks_exceptions timeout_cleanup abnormal_exit missing_evidence_rejected)
      add_test(NAME qemu_${case} COMMAND ${Python3_EXECUTABLE} -m unittest test_qemu.QemuAcceptance.test_${case})
      set_tests_properties(qemu_${case} PROPERTIES WORKING_DIRECTORY ${CMAKE_SOURCE_DIR}/tests/qemu TIMEOUT 25)
    endforeach()
    add_test(NAME elf_layout COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/check_elf.py)
    add_test(NAME gdb_verify COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/scripts/debug_verify.py)
    set_tests_properties(elf_layout PROPERTIES TIMEOUT 15)
    set_tests_properties(gdb_verify PROPERTIES TIMEOUT 45)
  endif()
  add_custom_command(TARGET cortex_m3_demo POST_BUILD COMMAND ${CMAKE_OBJCOPY} -O binary $<TARGET_FILE:cortex_m3_demo> ${CMAKE_BINARY_DIR}/cortex_m3_demo.bin COMMAND ${CMAKE_SIZE} $<TARGET_FILE:cortex_m3_demo>)
else()
  add_executable(host_smoke apps/host_smoke/main.c src/checksum.c)
  target_include_directories(host_smoke PRIVATE include)
  add_test(NAME host_smoke COMMAND host_smoke)
  include(cmake/stage2.cmake)
endif()
