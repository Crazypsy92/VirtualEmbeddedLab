cmake_minimum_required(VERSION 3.20)
project(VirtualEmbeddedLabStage5 NONE)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(VEL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(names environment official_ros host_mock protocol firmware_build uart e2e faults gdb resources)
set(scripts setup_stage5 stage5_ros_basic stage5_host_mock_test stage5_protocol_test stage5_build_firmware stage5_uart_test stage5_e2e stage5_faults stage5_gdb stage5_memory)
foreach(i RANGE 0 9)
  list(GET names ${i} name)
  list(GET scripts ${i} script)
  add_test(NAME "stage5_${name}" COMMAND "${Python3_EXECUTABLE}" "${VEL_ROOT}/scripts/${script}.py")
  set_tests_properties("stage5_${name}" PROPERTIES TIMEOUT 180 RUN_SERIAL TRUE WORKING_DIRECTORY "${VEL_ROOT}")
endforeach()
