cmake_minimum_required(VERSION 3.20)
project(VirtualEmbeddedLabStage6 NONE)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
function(gate name script)
 add_test(NAME "stage6_${name}" COMMAND "${Python3_EXECUTABLE}" "${ROOT}/scripts/${script}.py" ${ARGN})
 set_tests_properties("stage6_${name}" PROPERTIES TIMEOUT 900 RUN_SERIAL TRUE WORKING_DIRECTORY "${ROOT}")
endfunction()
gate(versions stage6_versions)
gate(host_build stage6_build_host)
gate(host_session stage6_host_test)
gate(dev_tools stage6_build_dev)
gate(private_newlib stage6_build_newlib)
gate(zephyr_configuration stage6_prepare_arm)
gate(arm_libraries stage6_build_arm_libs)
gate(minimal_arm_build stage6_build_firmware)
gate(arm_topic_session stage6_arm_runtime)
gate(arm_gdb stage6_gdb)
gate(interfaces stage6_build_interfaces)
gate(acquisition_build stage6_build_firmware --acquisition)
gate(acquisition_256 stage6_acquisition_test)
gate(fault_recovery stage6_faults)
gate(memory stage6_memory)
