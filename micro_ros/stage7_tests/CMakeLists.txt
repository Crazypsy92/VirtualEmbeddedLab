cmake_minimum_required(VERSION 3.20)
project(VirtualEmbeddedLabStage7 NONE)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
function(gate name label script)
 add_test(NAME "stage7_${name}" COMMAND "${Python3_EXECUTABLE}" "${ROOT}/scripts/${script}.py" ${ARGN})
 set_tests_properties("stage7_${name}" PROPERTIES LABELS "${label}" TIMEOUT 360 RUN_SERIAL TRUE WORKING_DIRECTORY "${ROOT}")
endfunction()
gate(environment baseline stage7_environment)
gate(recovery baseline stage7_recovery_test)
gate(stress baseline stage7_stress_test)
gate(faults baseline stage7_fault_test)
gate(gdb baseline stage7_gdb)
gate(opt_recovery optimized stage7_recovery_test --optimized)
gate(opt_stress optimized stage7_stress_test --optimized)
gate(opt_faults optimized stage7_fault_test --optimized)
gate(opt_gdb optimized stage7_gdb --optimized)
gate(memory resource stage7_memory)
