cmake_minimum_required(VERSION 3.20)
project(vel_micro_ros_host C)
find_package(ament_cmake REQUIRED)
find_package(rcl REQUIRED)
find_package(rclc REQUIRED)
find_package(rmw_microxrcedds REQUIRED)
find_package(std_msgs REQUIRED)
add_executable(vel_micro_ros_host main.c)
target_compile_features(vel_micro_ros_host PRIVATE c_std_11)
target_compile_options(vel_micro_ros_host PRIVATE -Wall -Wextra -Werror -g)
target_link_libraries(vel_micro_ros_host PRIVATE rcl::rcl rclc::rclc ${rmw_microxrcedds_LIBRARIES} ${std_msgs_TARGETS})
target_include_directories(vel_micro_ros_host SYSTEM PRIVATE ${rmw_microxrcedds_INCLUDE_DIRS})
