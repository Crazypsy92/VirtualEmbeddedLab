cmake_minimum_required(VERSION 3.20)
get_filename_component(VEL_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
list(APPEND DTS_ROOT ${VEL_ROOT})
find_package(Zephyr REQUIRED HINTS $ENV{ZEPHYR_BASE})
project(zephyr_micro_ros_firmware C)
set(UROS_INSTALL "${VEL_ROOT}/../build/workspaces/micro_ros_stage6_ws/arm_install" CACHE PATH "Pinned ARM micro-ROS install")
if(VEL_ACQUISITION)
 target_sources(app PRIVATE src/acquisition.c)
else()
 target_sources(app PRIVATE src/main.c)
endif()
target_sources(app PRIVATE ${VEL_ROOT}/drivers/sensor/vel_virtual_sensor/vel_virtual_sensor.c)
target_include_directories(app PRIVATE ${VEL_ROOT}/common ${UROS_INSTALL}/include)
file(GLOB UROS_INCLUDE_DIRS LIST_DIRECTORIES TRUE "${UROS_INSTALL}/include/*")
foreach(dir ${UROS_INCLUDE_DIRS})
 if(IS_DIRECTORY "${dir}")
  target_include_directories(app SYSTEM PRIVATE "${dir}")
 endif()
endforeach()
file(GLOB UROS_LIBRARIES "${UROS_INSTALL}/lib/*.a")
if(NOT UROS_LIBRARIES)
 message(FATAL_ERROR "Build real ARM micro-ROS libraries first")
endif()
zephyr_link_libraries(-Wl,--start-group ${UROS_LIBRARIES} -Wl,--end-group)
target_compile_options(app PRIVATE -Wall -Wextra -Werror)

target_compile_options(app PRIVATE -include ${VEL_ROOT}/../micro_ros/arm/platform_compat.h)
