cmake_minimum_required(VERSION 3.20)
project(VELZephyrAcceptance NONE)
enable_testing()
find_package(Python3 REQUIRED COMPONENTS Interpreter)
get_filename_component(VEL_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
foreach(case IN ITEMS environment hello sync kernel kernel_gdb sensor sensor_off missing_property sensor_gdb acq0 acq1 acq2 acq3 acq4 acq5 acq6 acquisition_gdb memory)
  add_test(NAME zephyr_${case} COMMAND ${Python3_EXECUTABLE} ${VEL_ROOT}/scripts/stage4_case.py ${case})
  set_tests_properties(zephyr_${case} PROPERTIES TIMEOUT 55)
endforeach()
